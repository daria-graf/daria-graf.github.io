# 10 get locators

```js
// 1. by tag name
cy.get('input');

// 2. by id
cy.get('#inputEmail');

// 3. by class name
cy.get('.input-full-width');

// 4. by attribute name
cy.get('[placeholder]');

// 5. by attribute name and value
cy.get('[placeholder="inputEmail"]');

// 6. by class value
cy.get('[class="input-full-width size-medium"]');

// 7. by tag name and attribute with value
cy.get('input[data-cy="inputEmail"]');

// 8. by two different attributes
cy.get('[placeholder="inputEmail"][type="email"]');

// 9. by tag name, attribute with value, id and class name
cy.get('input[placeholder="inputEmail"]#inputEmail.input-full-width');

// 10. the most recommended way by cypress - create our own attribute
cy.get('[data-cy="inputEmail"]');
```

# Find a DOM element by text

```js
// yield first element in document containing 'Sign in'
cy.contains('Sign in').click();
// with a regex and i flag for case insensitive
cy.contains(/sign in/i).click();
// or with matchCase attribute
cy.get('nav').contains('sign in', { matchCase: false });

// select a particular element, if more than one similar buttons
cy.contains('[status="warning"]', 'Sign in');
```

# Find a sibling element

Find() finds the elements inside a parent element.

```js
// find a button inside the form element which contains
// an input element with id="#inputEmail"
cy.get('#inputEmail').parents('form').find('button');

// find a button and check a checkbox after that
cy.get('#inputEmail')
    .parents('form')
    .find('button').
    .should('contain', 'Log in')
    .parents('form') // go up to the form element
    .find('checkbox')
    .click();
```

# Find a child element inside a parent

```js
// finds a card, which contains text 'Header'
// and inside the card finds an element with
// an attribute 'type="email"'
cy.contains('card', 'Header').find('[type="email"]');
```
